\chapter{Kết Luận, Đe Dọa Tính Hợp Lệ và Hướng Phát Triển}

\section{Tổng Kết Đóng Góp Khoa Học và Kỹ Thuật}
\label{sec:ch7_contributions}

Hệ thống \textbf{UTH Scientific Lakehouse, Data Mining \& Hybrid RAG} đã hoàn thành mục tiêu xây dựng một nền tảng hợp nhất từ thu hoạch tài liệu học thuật quy mô lớn đến suy luận hỏi đáp có kiểm soát trích dẫn, thể hiện qua bốn nhóm đóng góp cốt lõi:

\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Kiến trúc Lakehouse HTML-First và tách tầng lưu trữ hiệu năng cao}: Xây dựng thành công kho dữ liệu trên \textbf{71,927 bài báo phân biệt} (\textbf{1,043,594 vector} 768 chiều trên Qdrant tại ngày 04/11/2026), giữ tỷ lệ làm giàu HTML LaTeXML đạt \textbf{89.38\%} (2,825,871 công thức toán học ở lát cắt kiểm toán 13,000 bài), đồng thời cô lập văn bản dài khỏi bộ nhớ chỉ mục HNSW nhờ cặp \textbf{Qdrant 5-field payload} và \textbf{\texttt{ChunkStore} ZSTD Row-Group Min/Max} kèm bộ đệm LRU 20,000 đoạn (quyết định \texttt{D-058}).

  \item \textbf{Định lượng bốn quy luật miền dữ liệu khoa học qua OLAP DuckDB không máy chủ}: Phẫu thuật ma trận $11,659 \times 6$ bằng kiểm định D'Agostino $K^2$ ($p < 0.001$)~\cite{dagostino1973tests} và khung 3 Lăng kính đã chứng minh tính lệch đuôi dài cực đoan (\textit{Heavy Right Tail}, \texttt{words} có $S = +49.92, K = +3,879.80$), độ chênh lệch $7.0\times$ mật độ công thức giữa \texttt{stat.ML} (687.0 eq/bài) và \texttt{cs.CL} (98.3 eq/bài), tính bất đối xứng đa nhãn ở \texttt{cs.AI} và quan hệ phi tuyến hình chữ U ngược của \texttt{words\_x\_math\_density} ($r = +0.012$ so với $\rho = +0.143^{***}$), làm cơ sở toán học cho chuẩn hóa $\ln(1+x)$, chia đoạn cân bằng và phân cụm mềm.

  \item \textbf{Vận hành đồng bộ Bốn Trụ cột Khai phá Dữ liệu trong 234.57 giây toàn hồ}:
  \begin{itemize}
    \item \textit{Trụ cột 1}: Khai phá \textbf{45 luật kết hợp không trùng lặp} trên 61,319 giao dịch hợp lệ (\texttt{top lift = 34.940}) nhờ ngưỡng \texttt{min\_abs\_transactions = 25} và khử đối xứng \texttt{\_lift\_key}~\cite{han2000mining}.
    \item \textit{Trụ cột 2}: Triển khai \textbf{EM GMM Full-Covariance} kèm elip $2\sigma$ trên không gian chiếu Centered PCA 2D, lý giải hiện tượng đa tạp ngữ nghĩa liên thông (\texttt{Silhouette = 0.0490} trên 70,539 vector bài báo so với 0.3528 và 25.62\% bài liên ngành trên lát cắt kiểm toán 800 bài)~\cite{dempster1977maximum,rousseeuw1987silhouettes}.
    \item \textit{Trụ cột 3}: Hỗ trợ đồ thị chế độ kép (Citation DiGraph 8,892 nút / 86,295 cạnh so với mạng đồng tác giả trọng số phân đoạn $1/(N_p - 1)$) tính Damped PageRank ($\alpha = 0.85$) và phân cụm cộng đồng Louvain~\cite{page1999pagerank,blondel2008fast}.
    \item \textit{Trụ cột 4}: Theo dõi 48 chuyên ngành qua động lượng thị phần $\Delta\text{Share}$ bằng \texttt{dense\_rank()} và cô lập 1,000 bài báo dị biệt cấu trúc bằng Isolation Forest 5 chiều trên thang $\ln(1+x)$ chỉ trong 2.89 giây~\cite{liu2008isolation}.
  \end{itemize}

  \item \textbf{Đường ống Hybrid RAG có gia cố PageRank và Cổng Chứng cứ Hiệu chỉnh}: Khảo nghiệm cắt bỏ trên $N=120$ truy vấn chứng minh hiệu ứng cộng hưởng khi kết hợp Vector + Laya + RRF ($k=60$) + PageRank ($\mathrm{MRR}$ tăng +6.13\% lên 0.6076 và $\mathrm{Hit}@1$ tăng lên 55.0\%)~\cite{cormack2009reciprocal}. Đặc biệt, tinh chỉnh \texttt{Laya} ONNX bằng hàm mất mát \texttt{Soft-CE} và Temperature Scaling ($T^* = 3.0627$) đã kéo sai số hiệu chỉnh $\mathrm{ECE}$ giảm $5.72\times$ ($0.1900 \rightarrow 0.0332$), cắt giảm tỷ lệ chấp nhận nhầm bịa đặt từ 57.4\% xuống 1.5\%, đạt \textbf{$\mathrm{AUROC} = 0.986$} (chặn 50/50 câu bẫy âm tính ở \texttt{Retrieval v1}) và \textbf{$\mathrm{Hit}@5 = 0.98$}, \textbf{$\mathrm{MRR} = 0.8939$} ở đối sánh 3 chiều 150 mẫu; đồng thời chuyển đổi suy luận sang CUDA \texttt{node-llama-cpp} đạt \textbf{46.33 tok/s} (3.39 s E2E, tăng tốc $11.0\times$) trên GPU RTX 4060 8 GB.
\end{enumerate}

\section{Đe Dọa Tính Hợp Lệ và Hạn Chế Hiện Tại}
\label{sec:ch7_threats}

Nhằm bảo toàn tính trung thực khoa học, năm mối đe dọa tính hợp lệ (\textit{Threats to Validity}) và hạn chế kỹ thuật hiện hữu được nhận diện thẳng thắn:

\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Khoảng cách Serving giữa nhánh thử nghiệm \texttt{RAG\_TS} (\texttt{EXP-05}) và Backend FastAPI (\textit{Internal/External Validity})}: Các chỉ số đỉnh cao tại \texttt{EXP-05} ($\mathrm{AUROC} = 0.986$, $\mathrm{Hit}@5 = 0.94\text{--}0.98$, chặn 50/50 bẫy âm tính) được đo trên bộ chạy \texttt{RAG\_TS} sử dụng trọng số \texttt{Laya 0.4.27} ONNX tinh chỉnh, chỉ mục SQLite FTS5 BM25 độc lập và ngưỡng cứng Cổng Chứng cứ 0.6. Trong khi đó, dịch vụ sản xuất FastAPI hiện tại (\path{reranker_service.py}, \path{retrieval_service.py}) vẫn mặc định \path{cross-encoder/ms-marco-MiniLM-L-6-v2} (\texttt{max\_length = 256}), chấm điểm từ vựng nội tập thay vì BM25 toàn cục và chưa tự động từ chối ở ngưỡng 0.6 (cấu hình \texttt{prod} trên tập v1 chỉ đạt $\mathrm{MRR}@10 = 0.561$ so với 0.824 của \texttt{laya\_ft1\_hybrid}).

  \item \textbf{Sự khác biệt công thức PageRank và RRF giữa \texttt{retrieval\_service.py} và \texttt{EXP-04}}: Dịch vụ FastAPI chuẩn hóa điểm RRF ($k=60$) về đoạn $[0.70, 0.95]$ rồi nhân hệ số uy tín tuyến tính $(1 + 0.25 \cdot \mathrm{norm\_pr})$ ở cả bước tiền RRF và hậu RRF (Mục~\ref{sec:authority_multiplier}). Ngược lại, kịch bản \texttt{EXP-04} (\path{evaluate_120_ranking_5_scenarios.py}) sử dụng Weighted RRF (0.65 Vector / 1.35 Laya) kết hợp cộng biên $+0.025$ từ điểm PageRank nén logarit. Hai biến thể này cần hợp nhất về một công thức tham số hóa thống nhất trước khi mở rộng benchmark quy mô lớn.

  \item \textbf{Thiếu hụt Hydration trên một phần kho bài báo cũ (\textit{Construct Validity})}: Đánh giá DeepEval tại Mục~\ref{sec:ch5_stage_latency} cho thấy khi gộp thêm $\approx 38.9\%$ mẫu thiếu toàn văn ở tập $N=169$ (so với $N=105$ đã hydrate), \texttt{Context Recall} giảm từ 0.900 xuống 0.637. Cần hoàn tất chiến dịch bù đắp toàn văn trước khi đánh giá lại toàn hồ.

  \item \textbf{Nút thắt thời gian Đồ thị và giới hạn Từ vựng Luật Kết hợp}: Thuật toán Louvain và PageRank thuần Python trên \texttt{NetworkX} tiêu tốn \textbf{116.39 giây} (49.6\% tổng thời gian \texttt{MiningEngine}) cho 71,927 bài; khi kho dữ liệu tăng lên 500,000 bài, chi phí đồ thị đơn luồng sẽ vượt quá cửa sổ bảo trì ngắn. Đồng thời, bộ phân tích FP-Growth dựa trên từ điển tĩnh 25 cụm từ khóa \texttt{KEYWORD\_VOCAB} chưa thể tự động bắt kịp các khái niệm mới nổi (như \textit{Test-Time Compute}, \textit{MoE Routing} hay \textit{State Space Models}) nếu không cập nhật thủ công.

  \item \textbf{Tư thế bảo mật khóa fallback mặc định}: Dù \path{config.py} và \path{settings.py} dùng khóa đọc công khai cho Qdrant demo (\path{uth_guest_readonly_key_4b71a8e9d2c}), việc giữ chuỗi khóa fallback mặc định trong mã nguồn vẫn đòi hỏi quy trình tách biệt hoàn toàn cấu hình sản xuất và cấu hình trình diễn.
\end{enumerate}

\section{Hướng Phát Triển Tương Lai}
\label{sec:ch7_future_work}

Từ năm hạn chế thực tế trên, bốn hướng nâng cấp trực tiếp và khả thi cao được hoạch định cho giai đoạn tiếp theo:

\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Hợp nhất \texttt{Laya} ONNX tinh chỉnh ($T^* = 3.0627$) và Cổng Chứng cứ $\theta = 0.6$ vào FastAPI}: Tích hợp trực tiếp tệp ONNX tại \path{notebooks/laya_onnx_20261108-2053_seed0/} vào \texttt{RerankerService} qua \texttt{onnxruntime} (kèm tùy chọn GPU CUDA Execution Provider), phối hợp với chỉ mục BM25 toàn cục (SQLite FTS5 hoặc Qdrant Sparse Vectors) nhằm đưa dịch vụ FastAPI tiệm cận mốc \texttt{AUROC = 0.986} và $\mathrm{MRR}@10 = 0.827$ của \texttt{EXP-05}.

  \item \textbf{Hoàn thiện Hydration toàn phần và đồng bộ hóa dữ liệu}: Kích hoạt định kỳ \path{scripts/backfill_missing_papers.py} và \path{dags/dag_tier3_pdf_backfill.py} để đẩy tỷ lệ bài báo có đầy đủ đoạn văn trong \texttt{ChunkStore} tiệm cận 100\%, triệt tiêu nguyên nhân sụt giảm \texttt{Context Recall}.

  \item \textbf{Tăng tốc khai phá đồ thị và trích xuất khái niệm động}: Chuyển đổi lõi \texttt{ScientificGraphMiner} từ \texttt{NetworkX} thuần Python sang \texttt{python-igraph} (nền C) hoặc \texttt{cuGraph} (RAPIDS GPU) để nén thời gian tính PageRank và Louvain xuống dưới 5 giây; đồng thời nâng cấp \texttt{AssociationRuleMiner} từ \texttt{KEYWORD\_VOCAB} cố định sang trích xuất thuật ngữ học thuật động bằng KeyBERT hoặc LLM batch tagging định kỳ.

  \item \textbf{Cơ chế từ chối minh bạch và suy luận đa bước (Agentic Multi-Hop RAG)}: Kết nối trực tiếp tín hiệu \texttt{Evidence Gate < 0.6} từ \texttt{RerankerService} vào \texttt{RagService} để kích hoạt luồng từ chối có cấu trúc (hiển thị các bài liên quan gần nhất kèm xác suất hiệu chỉnh thấp thay vì ép LLM suy đoán), đồng thời tận dụng các cạnh trích dẫn trong đồ thị cho suy luận tổng hợp qua hai bước (\textit{1-hop citation follow-up}) với các câu hỏi so sánh chuyên sâu.
\end{enumerate}
