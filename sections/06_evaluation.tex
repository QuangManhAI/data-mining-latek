\chapter{Đánh Giá Thực Nghiệm Toàn Diện và Phân Tích Benchmark}
\label{chap:evaluation}

Để khẳng định tính đúng đắn của các quyết định thiết kế kiến trúc dữ liệu ở Chương~\ref{chap:data_preparation} và thuật toán khai phá kết hợp Hybrid RAG ở Chương~\ref{chap:modeling}, toàn bộ hệ thống được kiểm định định lượng qua năm chuỗi thực nghiệm có kiểm soát (\texttt{EXP-01} đến \texttt{EXP-05}). Các phép đo tách biệt rõ ràng từng tầng biến số: chiến lược chia đoạn học thuật (\texttt{AcademicChunker}), hiệu quả hợp nhất lai (Hybrid RRF + PageRank), hiệu chỉnh xác suất tái xếp hạng (\texttt{Laya} Cross-Encoder ONNX + Temperature Scaling), chất lượng sinh câu trả lời bằng giám khảo LLM (\texttt{DeepEval}), và thông lượng suy luận GPU thực tế trên NVIDIA RTX 4060.

Nhằm bảo đảm chuẩn mực liêm chính học thuật và đối chiếu đúng ngữ cảnh nguồn theo ma trận phân tách mốc dữ liệu tại Bảng~\ref{tab:snapshot_sources}, chương này tách bạch hai môi trường thực thi: (i) dịch vụ sản xuất Python FastAPI trên MinIO S3 + Qdrant + \texttt{ChunkStore} (\texttt{EXP-01}, \texttt{EXP-03}, \texttt{EXP-04}), và (ii) nhánh đánh giá độc lập \texttt{RAG\_TS} (NodeJS/TypeScript + SQLite BM25 + \texttt{Laya} ONNX) dùng để khảo nghiệm tinh chỉnh mô hình và Cổng Chứng cứ trên tập khóa cứng 150 truy vấn (\texttt{EXP-05}). Các số liệu thăm dò lịch sử trên LanceDB/Cloudflare R2 (05/10/2026) chỉ đóng vai trò bối cảnh tiền đề và không bị gộp chéo với phiên bản Qdrant hiện hành.

\section{Thiết Kế Khung Thực Nghiệm và Thước Đo Chuẩn Hóa}
\label{sec:ch5_framework}

Hệ thống đánh giá tổ chức bốn họ thước đo độc lập tương ứng với bốn tầng rủi ro của bài toán hỏi đáp khoa học:

\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Nhóm xếp hạng truy hồi (Information Retrieval Ranking)}: Hai thước đo cốt lõi đánh giá thứ hạng tài liệu đúng là Mean Reciprocal Rank ($\mathrm{MRR}$) và Normalized Discounted Cumulative Gain ($\mathrm{nDCG}@K$):
  \begin{equation}
    \mathrm{MRR} = \frac{1}{|Q|} \sum_{q \in Q} \frac{1}{\mathrm{rank}_q}, \qquad \mathrm{nDCG}@K = \frac{1}{|Q|} \sum_{q \in Q} \frac{1}{\mathrm{IDCG}_q@K} \sum_{i=1}^{K} \frac{2^{\mathrm{rel}_{q,i}} - 1}{\log_2(i + 1)}
    \label{eq:ch5_mrr_ndcg}
  \end{equation}
  Trong đó: $Q$ là tập truy vấn có đáp án chuẩn; $\mathrm{rank}_q$ là thứ hạng nhỏ nhất của bài báo hoặc đoạn văn đích trong danh sách trả về (bằng $\infty$ nếu không xuất hiện); $\mathrm{rel}_{q,i} \in \{0, 1, 2\}$ là nhãn liên quan theo từng cấp tại vị trí xếp hạng $i$; và $\mathrm{IDCG}_q@K$ là giá trị DCG lý tưởng khi sắp xếp tối ưu. Bên cạnh đó, hệ thống theo dõi $\mathrm{Hit}@K$ (tỷ lệ truy vấn có ít nhất một tài liệu đúng trong Top-$K$) và $\mathrm{Recall}@K$ (tỷ lệ tài liệu đích được phủ trong Top-$K$, tính chặt trên các câu hỏi đa bài báo).

  \item \textbf{Nhóm hiệu chỉnh xác suất và từ chối câu hỏi thiếu bằng chứng (Calibration \& Evidence Gate)}: Đo bằng diện tích dưới đường cong ROC ($\mathrm{AUROC}$), sai số hiệu chỉnh kỳ vọng $\mathrm{ECE}$ (\textit{Expected Calibration Error} --- độ lệch trung bình có trọng số giữa xác suất dự đoán và tần suất chính xác thực tế)~\cite{guo2017calibration}, chỉ số $\mathrm{Brier\ Score}$, và tỷ lệ từ chối đúng trên tập câu hỏi gài bẫy âm tính ($\mathrm{Negative\ Reject}@0.5$ và $@0.6$).

  \item \textbf{Nhóm đánh giá chất lượng câu trả lời LLM (\path{DeepEval})}: Sử dụng khung giám khảo tự động \path{DeepEval}~\cite{deepeval2024confident} vận hành cùng mô hình cục bộ \path{Qwen2.5-7B-Instruct}~\cite{qwen25report2024} trên năm chỉ số: \path{Context Recall} (mức độ ngữ cảnh chứa đủ luận điểm của đáp án chuẩn), \path{Context Precision} (độ ưu tiên thứ hạng ngữ cảnh hữu ích), \path{Citation Grounding} (tỷ lệ phát biểu kèm trích dẫn đúng nguồn), \path{Faithfulness} (mức trung thực không bịa đặt ngoài ngữ cảnh) và \path{Answer Relevancy} (mức bám sát trọng tâm câu hỏi).

  \item \textbf{Nhóm thông lượng phần cứng (Systems Performance)}: Đo thời gian sinh token đầu tiên ($\mathrm{TTFT}$), tốc độ giải mã tự hồi quy ($\mathrm{tok/s}$), phân vị độ trễ $p_{50}$/$p_{95}$ và thông lượng truy vấn ($\mathrm{QPS}$).
\end{enumerate}

\subsection{Kết Quả So Sánh 5 Kịch Bản Xếp Hạng trên 120 Truy Vấn}

Sau khi giải quyết nút thắt mất nội dung văn bản sau quyết định \texttt{D-058} bằng cơ chế thủy hóa \texttt{ChunkStore} từ Silver Parquet, kịch bản \path{benchmarks/evaluate_120_ranking_5_scenarios.py} thực thi nghiên cứu cắt bỏ (\textit{ablation study}) trên \textbf{120 câu hỏi học thuật} (92 truy vấn đơn bài và 28 truy vấn đa bài; chi tiết tại \path{docs/benchmarks/laya_pagerank_benchmark_report.md} và \path{logs/benchmarks/laya_5_scenarios_comparison_120q.json}). Cả năm kịch bản đều chấm điểm trên cùng một tập ứng viên Top-10 đã hydrate (\texttt{Hit@10 = 75.0\%}), tiến dần từ Vector đơn thuần đến hệ thống hợp nhất đầy đủ: Weighted RRF ($0.65$ Vector + $1.35$ Cross-Encoder \texttt{Laya})~\cite{cormack2009reciprocal} kết hợp nhân trọng số uy tín PageRank nén logarit (+0.025 trên điểm RRF, đối chiếu công thức xếp hạng tại Mục~\ref{sec:authority_multiplier} và Mục~\ref{sec:mining_rag_integration} trong Chương~\ref{chap:modeling}).

Kết quả tại Bảng~\ref{tab:laya_5_scenarios} khẳng định hai phát hiện thực nghiệm then chốt:

Cross-Encoder lọc nhiễu ngữ nghĩa ở cấp độ câu chữ cục bộ, RRF cân bằng thứ hạng tránh thiên vị logit cực đoan, còn PageRank phân giải các cặp hòa điểm dựa trên uy tín học thuật toàn cục.
  \item \textbf{Chi phí tra cứu PageRank gần như bằng không}: Việc ánh xạ từ bảng băm nạp sẵn trên RAM gồm 20,000 bài báo và 29,523 tác giả chỉ tiêu tốn thêm \textbf{+0.21 ms} mỗi truy vấn nhưng đóng góp +4.2 điểm phần trăm $\mathrm{Hit}@1$ ($50.8\% \rightarrow 55.0\%$) so với Vector + Laya RRF.
\end{itemize}

\section{Phân Tích Lỗi Trọng Yếu và Khoảng Cách Triển Khai}
\label{sec:ch5_postmortem}

Một báo cáo kỹ thuật minh bạch cần công bố đầy đủ các lỗi vận hành nghiêm trọng đã gặp cùng những khoảng cách còn tồn tại giữa nhánh đánh giá nghiên cứu và máy chủ API đang chạy:

\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Lỗi \texttt{BUG-01}: Sập bộ tách từ Rust do ký tự Unicode Surrogate} (\path{docs/bugs/BUG_01_TOKENIZER_SURROGATE_CRASH.md}):
  Trong quá trình nhúng văn bản hàng loạt bằng \texttt{nomic-embed-text-v1.5}, tiến trình worker đột ngột sập với ngoại lệ \texttt{PyO3 PanicException: no low surrogate in string} khi gặp các công thức HTML trích xuất bị vỡ cặp mã UTF-16 (\texttt{\textbackslash ud800--\textbackslash udfff}). Nguyên nhân gốc là kiểu \texttt{str} của Python cho phép tồn tại ký tự surrogate đơn lẻ, nhưng thư viện \texttt{tokenizers} viết bằng Rust ép chuyển đổi sang UTF-8 chuẩn nghiêm ngặt qua \texttt{.unwrap()} dẫn đến panic làm chết tiến trình. Nhóm đã khắc phục triệt để bằng bộ lọc làm sạch hai tầng tại cả \texttt{EmbedderService} (\path{backend/app/services/embedder_service.py}) và \texttt{NomicEmbedder} (\path{src/indexing/embedder.py}), kết hợp biểu thức chính quy loại bỏ dải surrogate và chuẩn hóa \texttt{encode("utf-8", errors="ignore").decode("utf-8")} trước mọi lệnh gọi tokenizer.

  \item \textbf{Lỗi \texttt{BUG-02} và Quyết định \texttt{D-058}: Gán sai nhãn \texttt{Methodology} và từ chối quá phòng thủ} (\path{docs/bugs/BUG_02_RAG_FALSE_METHODOLOGY_FALLBACK_AND_REFUSAL.md}):
  Khi đoạn văn truy hồi thuộc phần tóm tắt (\textit{Abstract}) thiếu tên tiểu mục, cả frontend và \texttt{RagService} từng mặc định gán nhãn sai là \texttt{"Methodology"} và prompt ép LLM từ chối trả lời toàn bộ nếu thiếu chi tiết vi mô. Lỗi này được xử lý bằng cách gán trung thực nhãn \texttt{[Section: Abstract]}, chuyển sang prompt \textit{Synthesis-First} tổng hợp nguyên lý toán học kèm nêu rõ giới hạn phạm vi, và tăng \texttt{RERANKER\_CANDIDATE\_K = 8}.

  \item \textbf{Khoảng cách Serving giữa Python FastAPI và nhánh \texttt{RAG\_TS} (\texttt{EXP-05})}:
  Trong khi hệ thống đánh giá \texttt{RAG\_TS} tại \texttt{EXP-05} đã sử dụng trọng số \texttt{Laya} ONNX tinh chỉnh kèm chỉ mục BM25 SQLite độc lập và Cổng Chứng cứ $\theta = 0.6$ để đạt \texttt{AUROC = 0.986}, dịch vụ sản xuất Python FastAPI hiện hành (\path{backend/app/services/reranker_service.py}) vẫn cấu hình mặc định \path{cross-encoder/ms-marco-MiniLM-L-6-v2} (\texttt{max\_length = 256}) và chấm điểm từ vựng nội tập thay vì gọi trực tiếp tệp \texttt{Laya} ONNX tinh chỉnh với $T^* = 3.0627$. Việc hợp nhất mô hình \texttt{Laya} ONNX vào \texttt{RerankerService} là hạng mục nâng cấp ưu tiên hàng đầu được đề xuất ở Chương~\ref{chap:conclusion}.

  \item \textbf{Điểm nghẽn Hydration trên tập mở rộng ($N=105$ vs $N=169$)}:
  Như Bảng~\ref{tab:deepeval_hydrated} đã chỉ ra, khoảng 38.9\% bài báo trong tập kiểm thử mở rộng chưa có đoạn văn đầy đủ trong \texttt{ChunkStore}, kéo chỉ số \texttt{Context Recall} từ 0.900 xuống 0.637. Khoảng trống dữ liệu này đòi hỏi vận hành định kỳ các kịch bản bù đắp toàn văn (\path{scripts/backfill_missing_papers.py}) được trình bày trong Chương~\ref{chap:deployment}.
\end{enumerate}
